// ============================================================ // N8N CODE NODE // Mode: Run Once for Each Item // // Chức năng: // 1. Parse payload từ multipart/form-data // 2. Giữ nguyên các dữ liệu cũ // 3. Thêm uid_form // 4. Thêm conversation_id // 5. Lấy thời gian min/max // 6. Random wait_seconds // 7. Giữ nguyên binary file_anh / file_video // ============================================================ // ============================================================ // 1. LẤY BODY // ============================================================ const body = $json.body || {}; // ============================================================ // 2. TÁCH PAYLOAD // // Khi HTML gửi multipart/form-data: // body.payload thường là STRING: // // '{"noi_dung":"abc","thoi_gian_cho":{...}}' // // Ta chuyển nó thành OBJECT. // ============================================================ let payload = body.payload ?? {}; if (typeof payload === 'string') { try { payload = JSON.parse(payload); } catch (error) { console.log( '[PAYLOAD PARSE ERROR]', error.message ); payload = {}; } } // Đảm bảo luôn là object if ( !payload || typeof payload !== 'object' || Array.isArray(payload) ) { payload = {}; } // ============================================================ // 3. LẤY uid_form // // Ưu tiên theo thứ tự: // payload // root json // body // ============================================================ const uidForm = payload.uid_form ?? $json.uid_form ?? body.uid_form ?? null; // ============================================================ // 4. LẤY conversation_id // ============================================================ const conversationId = payload.conversation_id ?? $json.conversation_id ?? body.conversation_id ?? null; // ============================================================ // 5. THÊM uid_form + conversation_id VÀO PAYLOAD // ============================================================ payload = { ...payload, uid_form: uidForm, conversation_id: conversationId }; // ============================================================ // 6. LẤY CẤU HÌNH THỜI GIAN // ============================================================ let min = Number( payload?.thoi_gian_cho?.min_seconds ?? 50 ); let max = Number( payload?.thoi_gian_cho?.max_seconds ?? 100 ); // ============================================================ // 7. KIỂM TRA MIN / MAX // ============================================================ if (!Number.isFinite(min)) { min = 50; } if (!Number.isFinite(max)) { max = 100; } // Làm tròn xuống min = Math.floor(min); max = Math.floor(max); // Không cho số âm min = Math.max(0, min); max = Math.max(0, max); // Nếu người dùng nhập: // // min = 100 // max = 50 // // thì tự đảo lại if (min > max) { const temp = min; min = max; max = temp; } // ============================================================ // 8. CẬP NHẬT THỜI GIAN VÀO PAYLOAD // ============================================================ payload.thoi_gian_cho = { ...(payload.thoi_gian_cho || {}), mode: payload?.thoi_gian_cho?.mode ?? 'random', fixed_seconds: payload?.thoi_gian_cho?.fixed_seconds ?? null, min_seconds: min, max_seconds: max }; // ============================================================ // 9. RANDOM THỜI GIAN CHỜ // // Ví dụ: // min = 50 // max = 100 // // kết quả có thể: // 57 // 83 // 99 // ... // ============================================================ const waitSeconds = Math.floor( Math.random() * (max - min + 1) ) + min; // ============================================================ // 10. THÔNG TIN BINARY // // HTML multipart gửi: // // $binary.file_anh // $binary.file_video // // KHÔNG chuyển Base64. // ============================================================ const hasImage = !!$binary?.file_anh; const hasVideo = !!$binary?.file_video; // ============================================================ // 11. OUTPUT // // QUAN TRỌNG: // // binary: $binary // // để ảnh/video tiếp tục đi qua: // // Code // ↓ // Loop Over Items // ↓ // Wait // ↓ // node gửi Zalo // ↓ // quay lại Loop // ============================================================ return { json: { // Giữ toàn bộ JSON cũ ...$json, // ====================================== // BODY // ====================================== body: { ...body, // payload đã được đổi từ string -> object payload: payload }, // ====================================== // PAYLOAD TÁCH RIÊNG // ====================================== payload: payload, // ====================================== // ĐƯA RA ROOT ĐỂ NODE SAU DỄ LẤY // ====================================== uid_form: uidForm, conversation_id: conversationId, // ====================================== // THỜI GIAN RANDOM // ====================================== wait_seconds: waitSeconds, // ====================================== // DEBUG FILE // ====================================== has_image: hasImage, has_video: hasVideo }, // ========================================================== // CỰC KỲ QUAN TRỌNG // GIỮ FILE QUA CÁC NODE SAU // ========================================================== binary: $binary || {} };
